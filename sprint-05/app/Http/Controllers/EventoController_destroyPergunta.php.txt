<?php
// Cole este método dentro do seu EventoController (ou PerguntaController).

use App\Models\Pergunta;

public function destroyPergunta(Pergunta $pergunta)
{
    $this->authorize('delete', $pergunta);

    $pergunta->delete();

    return back()->with('success', 'Pergunta excluída com sucesso.');
}
